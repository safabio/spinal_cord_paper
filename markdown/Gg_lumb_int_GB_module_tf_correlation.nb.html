spearman

# Calculate Pearson correlation between each TF and module (apply takes 100 seconds)
sm_correlations <- apply(tf_exp, 
                         MARGIN = 2, 
                         function(x) apply(avg_mod_exp,
                                           MARGIN = 2,
                                           function(y) cor(x, y, method = "spearman")))

# Convert correlations to a data frame for easier analysis
